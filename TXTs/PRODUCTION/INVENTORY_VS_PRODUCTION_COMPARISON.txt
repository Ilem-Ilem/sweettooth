============================================

Key Components:
  ✅ Audit modal properties
  ✅ Opens modal on create/edit/delete
  ✅ Captures reason before action
  ✅ Creates ApprovalAuditRequest
  ✅ Supervisor review workflow
  ✅ Change applied only after approval

Code Structure:
─────────────
class Items extends BaseComponent
{
    // Audit Modal Properties
    public bool $showAuditModal = false;
    public ?string $auditAction = null;           // create|edit|delete
    public string $auditReason = '';
    public ?int $pendingItemId = null;
    public array $pendingItemData = [];

    // Create Item
    public function openCreateModal()
    {
        $this->reset(['item', 'itemId', 'errors']);
        $this->showItemModal = true;
    }

    // Edit Item
    public function openEditModal($id)
    {
        $this->itemId = $id;
        $this->item = Item::find($id)?->toArray();
        $this->showItemModal = true;
    }

    // Save (Create or Edit)
    public function save()
    {
        // Validate form data
        $this->validate($this->itemValidationRules());

        // Prepare data
        $this->prepareItemData();

        // Instead of directly saving, open audit modal
        $this->auditAction = $this->itemId ? 'edit' : 'create';
        $this->pendingItemId = $this->itemId;
        $this->pendingItemData = $this->item;
        $this->showAuditModal = true;  // ← SHOW AUDIT MODAL
    }

    // Submit Audit Request
    public function submitAuditRequest()
    {
        // Validate reason
        $this->validate([
            'auditReason' => 'required|string|min:10|max:500',
        ]);

        // Create approval request
        ApprovalAuditRequest::create([
            'user_id' => auth()->id(),
            'model_type' => 'App\Models\Item',
            'model_id' => $this->pendingItemId ?? 0,
            'action' => $this->auditAction,
            'reason' => $this->auditReason,
            'status' => 'pending',
        ]);

        // Close modal and show success
        $this->closeAuditModal();
        $this->toast()->success('Approval request submitted')->send();
    }

    // Delete Item
    public function delete($id)
    {
        $this->pendingItemId = $id;
        $this->auditAction = 'delete';
        $this->showAuditModal = true;  // ← SHOW AUDIT MODAL
    }

    // Close Audit Modal
    private function closeAuditModal()
    {
        $this->showAuditModal = false;
        $this->auditReason = '';
        $this->auditAction = null;
        $this->pendingItemId = null;
        $this->pendingItemData = [];
    }
}

Workflow:
─────────
User Action (Create/Edit/Delete)
    ↓
Form validation
    ↓
Audit Modal Opens ("Reason?" prompt)
    ↓
User enters reason (10+ chars)
    ↓
submitAuditRequest() called
    ↓
ApprovalAuditRequest created (status=pending)
    ↓
Modal closes, success message shown
    ↓
✅ Supervisor reviews and approves
    ↓
✅ Changes applied after approval

class Products extends BaseComponent
{

        $this->product = Product::find($id)?->toArray();
        $this->productId = $id;
        $this->showItemModal = true;
    }

    {
        $this->validate($this->rules());

            // ❌ DIRECT CREATE - NO APPROVAL PROCESS

        }
        
        // ❌ No ApprovalAuditRequest created
        // ✅ Audit log created by ProductionAuditService (minimal info)
        $this->toast()->success('Product saved')->send();

    {
        $this->dialog()->confirm([
            'title' => 'Delete Product',
            'description' => 'Are you sure?',
            'acceptAction' => 'confirmedDelete',

    }

    {
        // ❌ DIRECT DELETE - NO APPROVAL PROCESS

    // ❌ NO submitAuditRequest() method
    // ❌ NO closeAuditModal() method
    // ❌ NO audit modal properties
