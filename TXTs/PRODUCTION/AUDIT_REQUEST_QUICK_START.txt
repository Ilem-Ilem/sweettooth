CODE SNIPPETS TO ADD

SNIPPET 1: Add Properties (Copy to each file)
============================================
// Audit modal for approval requests
public bool $showAuditModal = false;
public ?string $auditAction = null;          // create|edit|delete
public string $auditReason = '';              // User-provided reason
public ?int $pendingItemId = null;            // Product/Recipe ID pending action
public array $pendingItemData = [];           // Data to save on approval


SNIPPET 2: Update save() Method (Products.php example)
=======================================================
public function save()
{
    $this->validate($this->rules());
    
    // Instead of saving directly, open audit modal first
    $this->auditAction = $this->productId ? 'edit' : 'create';
    $this->pendingItemId = $this->productId;
    $this->pendingItemData = $this->getProductData();  // Gather data
    $this->showAuditModal = true;  // SHOW MODAL
}


SNIPPET 3: Update delete() Method
==================================
public function delete($id)
{
    $this->pendingItemId = $id;
    $this->auditAction = 'delete';
    $this->showAuditModal = true;  // SHOW MODAL INSTEAD OF DIALOG
}


SNIPPET 4: Add submitAuditRequest() Method
===========================================
public function submitAuditRequest()
{
    $this->validate([
        'auditReason' => 'required|string|min:10|max:500',
    ]);

    ApprovalAuditRequest::create([
        'user_id' => auth('employees')->id(),
        'model_type' => 'App\Models\Product',  // Or Recipe
        'model_id' => $this->pendingItemId,
        'action' => $this->auditAction,
        'reason' => $this->auditReason,
        'status' => 'pending',
    ]);

    $this->closeAuditModal();
    $this->toast()->success('Approval request submitted')->send();
}


SNIPPET 5: Add closeAuditModal() Method
========================================
private function closeAuditModal()
{
    $this->showAuditModal = false;
    $this->auditReason = '';
    $this->auditAction = null;
    $this->pendingItemId = null;
    $this->pendingItemData = [];
}


SNIPPET 6: Add to View (Include audit modal)
=============================================
@livewire('branch-dashboard.production.partials.audit-modal', [
    'showAuditModal' => $showAuditModal,
    'auditAction' => $auditAction,
    'auditReason' => $auditReason,
])

Or in Livewire component directly:
<x-audit-modal-production 
    :show="$showAuditModal"
    :action="$auditAction"
    wire:model="auditReason"
/>
