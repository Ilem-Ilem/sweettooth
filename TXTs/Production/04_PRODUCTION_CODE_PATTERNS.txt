    public function approveCallback($callbackId)
    {
        try {
            DB::beginTransaction();

            $callback = ProductDispatchCallback::find($callbackId);
            if (!$callback) {
                $this->toast()->error('Callback not found.')->send();
                return;
            }

            // Get current authenticated actor (User or Employee)
            $actor = current_actor();  // ← KEY PATTERN
            if (!$actor) {
                $this->toast()->error('No authenticated actor found.')->send();
                return;
            }

            // Pass actor object to model method
            $callback->approve($actor);  // ← Pass object, not ID

            DB::commit();
            $this->toast()->success('Callback approved successfully!')->send();
            $this->dispatch('$refresh');

        } catch (\Exception $e) {
            DB::rollBack();
            $this->toast()->error('Failed to approve callback: ' . $e->getMessage())->send();
        }
    }

    class ProductionCallback extends Model
    {
        // Store actor with type
        protected $fillable = [
            'recorded_by_id',       // UUID
            'recorded_by_type',     // String: App\Models\Employee or App\Models\User
            'approved_by_id',
            'approved_by_type',
        ];

        // Define morphTo relationship
        public function recordedBy(): MorphTo
        {
            return $this->morphTo('recorded_by', 'recorded_by_type', 'recorded_by_id');
        }

        public function approvedBy(): MorphTo
        {
            return $this->morphTo('approved_by', 'approved_by_type', 'approved_by_id');
        }

        // Use in methods with actor parameter
        public function approve($actor = null): bool
        {
            $actor = $actor ?? current_actor();  // ← Fallback to current if not provided
            if (!$actor) {
                throw new RuntimeException('No authenticated actor found');
            }

            $this->update([
                'approved_by_id' => $actor->id,
                'approved_by_type' => get_class($actor),  // ← Store full class name
                'approved_at' => now(),
            ]);

            return true;
        }
    }

❌ WRONG:

    // Only storing ID, missing type
    protected $fillable = ['approved_by'];
    
    public function approvedBy(): BelongsTo
    {
        return $this->belongsTo(Employee::class, 'approved_by');
    }

PATTERN 7: QUANTITY VALIDATION
===============================

✅ CORRECT:

    public function validateQuantity(): bool
    {
        // Different rules for different types
        if ($this->isRawMaterial()) {
            // Raw materials may have been partially used
            return true;  // No strict validation
        }

        if ($this->isFinishedProduct()) {
            if (!$this->product_id) {
                throw new InvalidArgumentException('Missing product_id');
            }

            $dailyProduce = DailyProduce::whereHas('recipe', function ($q) {
                $q->where('product_id', $this->product_id);
            })
            ->where('shift_id', $this->shift_id)
            ->lockForUpdate()  // ← Prevent race condition during concurrent checks
            ->first();

            if (!$dailyProduce) {
                throw new RuntimeException('DailyProduce not found');
            }

            if ($this->quantity > $dailyProduce->produced_quantity) {
                throw new ValidationException(
                    "Callback quantity ({$this->quantity}) exceeds produced ({$dailyProduce->produced_quantity})"
                );
            }
        }

        return true;
    }

❌ WRONG:

    public function validateQuantity(): bool
    {
        // No differentiation between types
        $maximum = 100;  // Arbitrary
        if ($this->quantity > $maximum) {
            throw new ValidationException('Too much');
        }
        return true;
    }

Function: current_actor()
Location: app/Helpers/BranchHelper.php

    function current_actor(): User|Employee|null
    {
        if (auth()->check()) {
            return auth()->user();  // Check default guard first
        }
        if (auth('employees')->check()) {
            return auth('employees')->user();  // Check employees guard
        }
        return null;
    }

Usage:
    $actor = current_actor();
    if ($actor) {
        // Use $actor->id and get_class($actor)
    }


Function: get_class()
Built-in PHP function

    get_class($actor);  // Returns "App\Models\Employee" or "App\Models\User"

Usage with polymorphic:
    'recorded_by_type' => get_class($actor)

DATABASE COLUMN NAMING CONVENTION
==================================

✅ CORRECT:

    Schema::create('productioncallbacks', function (Blueprint $table) {
        $table->id();
        
        // Polymorphic columns (always id + type pair)
        $table->uuid('recorded_by_id');       // FK ID
        $table->string('recorded_by_type');   // Model class name
        
        $table->uuid('approved_by_id')->nullable();
        $table->string('approved_by_type')->nullable();
        
        // Other columns...
        $table->timestamps();
    });

Naming rule:
    - Use snake_case (recorded_by_id, not recordedById)
    - Always pair _id with _type
    - _type stores class name (App\Models\Employee)
    - _id stores the primary key value

Accessing in model:
    $this->morphTo('recorded_by', 'recorded_by_type', 'recorded_by_id')
                  ↑ relation name  ↑ type column    ↑ id column

    public function test_production_callback_stores_actor_type()
    {
        $employee = Employee::factory()->create();
        
        $callback = ProductionCallback::create([
            'shift_id' => Shift::factory()->create()->id,
            'source_type' => 'raw_material_from_stock',
            'item_id' => Item::factory()->create()->id,
            'recorded_by_id' => $employee->id,
            'recorded_by_type' => Employee::class,  // ← Store type
            'quantity' => 5,
            'reason' => 'damaged',
            'status' => 'pending',
        ]);

        $this->assertEquals(Employee::class, $callback->recorded_by_type);
        $this->assertInstanceOf(Employee::class, $callback->recordedBy);
        $this->assertEquals($employee->id, $callback->recordedBy->id);
    }
