Components Properties to Add
php
public $viewMode = 'all'; // or 'top', 'category', 'low', etc.
// Add to queryString
protected $queryString = [
    'viewMode',

];
// Methods for each tab
public function getViewData()
{
    return match($this->viewMode) {
        'top' => $this->getTopItems(),
        'category' => $this->getCategoryData(),
        'low' => $this->getLowStockItems(),
        default => $this->getAllItems(),
    };
}

Database Queries to Add
StockMovementAnalytics
php
// Movement type distribution
private function getMovementTypeDistribution()
{
    return StockMovement::selectRaw('type, COUNT() as count')
        ->whereHas('stock', fn($q) => $q->where('branchid', $branchId))
        ->whereBetween('movementdate', [$dateFrom, $dateTo])
        ->groupBy('type')
        ->get();
}
// Hourly breakdown
private function getHourlyBreakdown()
{
    return StockMovement::selectRaw('HOUR(movementdate) as hour, COUNT() as count')
        ->whereHas('stock', fn($q) => $q->where('branchid', $branchId))
        ->whereBetween('movementdate', [$dateFrom, $dateTo])
        ->groupBy('hour')
        ->orderBy('hour')
        ->get();
}
// Source breakdown (by reference type)
private function getSourceBreakdown()
{
    return StockMovement::selectRaw('referencetype, COUNT() as count')
        ->whereHas('stock', fn($q) => $q->where('branchid', $branchId))
        ->whereBetween('movementdate', [$dateFrom, $dateTo])
        ->groupBy('referencetype')
        ->get();
}
RequestDispatchAnalytics
php
// Fulfillment by department
private function getFulfillmentByDepartment()
{
    return ItemRequest::selectRaw(
        'departmentid,
         COUNT() as total,
         COUNT(CASE WHEN status = "completed" THEN 1 END) as completed,
         COUNT(CASE WHEN status = "pending" THEN 1 END) as pending'
    )
    ->groupBy('departmentid')
    ->with('department')
    ->get();
}
// Average fulfillment time
private function getAverageFulfillmentTime()
{
    return ItemRequest::selectRaw(
        'departmentid,
         AVG(TIMESTAMPDIFF(MINUTE, createdat, completedat)) as avgminutes'
    )
    ->whereNotNull('completedat')
    ->groupBy('departmentid')
    ->get();
}
StockLevelAnalytics
php
// Health status breakdown
private function getHealthStatusBreakdown()
{
    $reorderPoint = 50; // Adjust based on config
    
    return Stock::selectRaw(
        'CASE 
            WHEN quantityavailable <= 0 THEN "outofstock"
            WHEN quantityavailable <= ? THEN "low"
            ELSE "healthy"
         END as status,
         COUNT() as count'
    )
    ->setBindings([$reorderPoint])
    ->groupBy('status')
    ->get();
}

Table Pattern
blade
<table class="w-full text-sm">

            <th class="px-4 py-2 text-left">Column 1</th>

            </tr>

            <tr>
                <td colspan="2" class="px-4 py-8 text-center text-zinc-500">No data</td>
            </tr>
        @endforelse
    </tbody>
</table>
