Adding Approval to Any Component
Step 1: Add Properties to Component
php
// In component class
public $showAuditModal = false;
public $auditReason = '';
public $auditAction = null;
public $pendingActionData = []; // Or specific field names
public $pendingItemId = null;   // For single-item actions
Step 2: Create Request Method
php
public function submitActionApprovalRequest()
{
    $this->validate([
        'auditReason' => 'required|string|min:10|max:500',
    ]);
    try {
        $actor = Auth::guard('employees')->user();
        // Call approval service
        $request = InventoryApprovalService::requestAction(
            $actor,
            $this->pendingItemId, // or data array
            $this->auditReason
        );
        // Verify creation
        if (!$request || !$request->id) {
            throw new \Exception('Failed to create approval request');
        }
        // Success feedback
        $this->toast()->success('Request submitted! (Request ID: ' . $request->id . ')')->send();
        $this->closeAuditModal();
    } catch (\Exception $e) {
        $this->toast()->error('Failed: ' . $e->getMessage())->send();
    }
}

Step 4: Add Method to InventoryApprovalService
php
public static function requestAction(
    Employee $requester,
    int $itemId, // or array $data
    string $reason
): ApprovalAuditRequest {
    $request = ApprovalAuditRequest::create([
        'branchid' => $requester->branchid,
        'requesterid' => $requester->id,
        'requestertype' => Employee::class,
        'action' => 'actionname:' . $itemId,
        'description' => $reason,
        'payload' => ['itemid' => $itemId], // Complete data
        'status' => 'pending',
    ]);
    AuditService::log(
        $requester,
        'create',
        $request,
        "Requested action for item. Reason: {$reason}",
        'pending'
    );
    return $request;
}
public static function executeAction(
    ApprovalAuditRequest $request,
    Employee $approver
): void {
    DB::transaction(function () use ($request, $approver) {
        $payload = $request->payload;
        $itemId = $payload['itemid'];
        // Do the actual action
        Model::find($itemId)->update($payload);
        // Mark approved
        $request->update([
            'approverid' => $approver->id,
            'approvertype' => Employee::class,
            'status' => 'approved',
            'approvedat' => now(),
        ]);
    });
}

Common Patterns
Pattern 1: Simple Delete
php
public function delete($id)
{
    $item = Model::findOrFail($id);
    if (issuperadmin()) {
        // Execute immediately
        $item->delete();
        AuditService::log(/.../);
        return;
    }
    // Request approval
    $this->pendingItemId = $id;
    $this->auditAction = 'deleteaction';
    $this->showAuditModal = true;
}
public function submitDeletionRequest()
{
    $request = InventoryApprovalService::requestItemDeletion(
        Auth::guard('employees')->user(),
        $this->pendingItemId,
        $this->auditReason
    );
    // ... handle result
}
Pattern 2: Form Save with Audit
php
public function save()
{
    $this->validate();
    if (issuperadmin()) {
        // Direct execution
        $this->executeImmediateSave();
        return;
    }
    // Store for approval
    $this->pendingActionData = [
        'field1' => $this->field1,
        'field2' => $this->field2,
    ];
    $this->auditAction = 'createaction';
    $this->showAuditModal = true;
}
public function submitAuditRequest()
{
    $request = InventoryApprovalService::requestActionCreation(
        Auth::guard('employees')->user(),
        $this->pendingActionData,
        $this->auditReason
    );
    // ... handle result
}

Debugging
Request Not Creating?
Check component has all required properties:
   php
   public $showAuditModal = false;
   public $auditReason = '';
   public $auditAction = null;
   
Check Service method returns request:
   php
   return $request;  // Must have this at end
   
Check Livewire closes modals:
   php
   $this->closeAuditModal();  // Must call this
   
Check database:
   sql
   SELECT  FROM approvalauditrequests ORDER BY id DESC LIMIT 5;
   
Request Not Visible in Approvals?
Check approvalauditrequests has the record
Check branchid matches current branch
Check status is pending (not approved/rejected)
Refresh the Approvals page
Approval Not Executing?
Check InventoryApprovals.php has handler for action type
Check Service has execute method
Check executeMethod is called in approveRequest()
Check database transaction completes
