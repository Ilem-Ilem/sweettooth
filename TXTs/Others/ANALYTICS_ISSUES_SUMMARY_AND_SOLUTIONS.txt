Issue 1: Shift & Department Filters Non-Functional
Current Problem
Location: app/Livewire/BranchDashboard/Analytics/StockMovementAnalytics.php (Lines 167-168)

->when($this->filterDepartment, fn ($q) => $q->whereHasMorph('reference', ['App\Models\ItemRequest'], fn ($sq) => $sq->where('departmentid', $this->filterDepartment)))
What's wrong:
Only filters movements with referencetype = 'App\Models\ItemRequest'
Excludes movements from Purchases (referencetype = 'App\Models\Purchase')
Excludes manual adjustments (referencetype = null)
Results in incomplete/incorrect filtered data
Impact:
Users see incomplete results when filtering by shift/department
Data analysis is inaccurate
Reports are unreliable
Solution: Add Shift/Department Denormalization

Issue 2: PurchaseAnalytics Missing Date Validation
Current Problem
Location: app/Livewire/BranchDashboard/Analytics/PurchaseAnalytics.php (Lines 18-28)

Impact:
Silent failures with invalid dates
Confusing user experience
Potential performance issues with large ranges
Solution: Add Date Validation
File: app/Livewire/BranchDashboard/Analytics/PurchaseAnalytics.php
Add these methods:
php
public function updatedDateFrom()
{
    $this->validateDateRange();
    $this->resetPage();
    $this->dispatch('chartsUpdated', [
        'trendData' => $this->getPurchaseTrendData(),
        'supplierAnalysis' => $this->getSupplierAnalysis(),
        'costBreakdown' => $this->getCostBreakdown(),
        'topItems' => $this->getTopPurchasedItems()->map(function($item) {
            return [
                'name' => $item->item->name,
                'totalcost' => $item->totalcost,
                'totalquantity' => $item->totalquantity,
                'uom' => $item->item->uom
            ];
        })->toArray()
    ]);
}
public function updatedDateTo()
{
    $this->validateDateRange();
    $this->resetPage();
    $this->dispatch('chartsUpdated', [
        'trendData' => $this->getPurchaseTrendData(),
        'supplierAnalysis' => $this->getSupplierAnalysis(),
        'costBreakdown' => $this->getCostBreakdown(),
        'topItems' => $this->getTopPurchasedItems()->map(function($item) {
            return [
                'name' => $item->item->name,
                'totalcost' => $item->totalcost,
                'totalquantity' => $item->totalquantity,
                'uom' => $item->item->uom
            ];
        })->toArray()
    ]);
}
private function validateDateRange()
{
    $from = \Carbon\Carbon::parse($this->dateFrom);
    $to = \Carbon\Carbon::parse($this->dateTo);
    
    if ($from->greaterThan($to)) {
        // Swap dates
        $temp = $this->dateFrom;
        $this->dateFrom = $this->dateTo;
        $this->dateTo = $temp;
        
        session()->flash('warning', 'Date range was automatically corrected.');
    }
    
    // Warn if range > 1 year
    if ($from->diffInDays($to) > 365) {
        session()->flash('warning', 'Large date ranges may impact performance.');
    }
}
Update mount() to add validation:
php
public function mount()
{
    $this->dateFrom = now()->subDays(90)->format('Y-m-d');
    $this->dateTo = now()->format('Y-m-d');
    $this->validateDateRange();
}

$branchId = @Auth::guard('employees')->user()->branchid ?? request()->get('bid');

$branchId = Auth::guard('employees')->user()?->branchid ?? request()->get('bid');

private function getBranchId()

$branchId = $this->getBranchId();

private function getAnalyticsSummary($branchId)
{
    $dateFrom = Carbon::parse($this->dateFrom)->startOfDay();
    $dateTo = Carbon::parse($this->dateTo)->endOfDay();
    $baseQuery = StockMovement::query()
        ->whereHas('stock', fn ($q) => $q->where('branchid', $branchId))
        ->whereBetween('movementdate', [$dateFrom, $dateTo]);
    $filteredQuery = clone $baseQuery;
    $this->applyFiltersToQuery($filteredQuery);
    // OPTIMIZED: Single query for period aggregations
    $periodData = $filteredQuery->selectRaw('
        SUM(CASE WHEN type IN ("in","return") THEN quantity ELSE 0 END) as stockin,
        ABS(SUM(CASE WHEN type IN ("out","damaged","transfer") THEN quantity ELSE 0 END)) as stockout,
        COUNT(CASE WHEN type = "transfer" THEN 1 END) as transfers,
        COUNT() as totalmovements,
        COUNT(CASE WHEN type = "adjustment" THEN 1 END) as adjustments,
        ABS(SUM(CASE WHEN type = "damaged" THEN quantity ELSE 0 END)) as damaged
    ')->first();
    // OPTIMIZED: Single query for today
    $todayQuery = StockMovement::query()
        ->whereHas('stock', fn ($q) => $q->where('branchid', $branchId))
        ->whereDate('movementdate', today());
    $todayData = $todayQuery->selectRaw('
        SUM(CASE WHEN type IN ("in","return") THEN quantity ELSE 0 END) as stockin,
        ABS(SUM(CASE WHEN type IN ("out","damaged","transfer") THEN quantity ELSE 0 END)) as stockout,
        COUNT(CASE WHEN type = "transfer" THEN 1 END) as transfers,
        COUNT() as totalmovements
    ')->first();
    // ... rest of analytics ...
    return [
        'today' => [
            'stockin' => $todayData->stockin ?? 0,
            'stockout' => $todayData->stockout ?? 0,
            'transfers' => $todayData->transfers ?? 0,
            'totalmovements' => $todayData->totalmovements ?? 0,
        ],
        'period' => [
            'stockin' => $periodData->stockin ?? 0,
            'stockout' => $periodData->stockout ?? 0,
            'transfers' => $periodData->transfers ?? 0,
            'totalmovements' => $periodData->totalmovements ?? 0,
            'adjustments' => $periodData->adjustments ?? 0,
            'damaged' => $periodData->damaged ?? 0,
        ],
        // ... rest of data ...
    ];
}
