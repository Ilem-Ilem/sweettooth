public function save()
{
    $this->validate($this->itemValidationRules());
    $itemData = [
        'branchid' => $this->getBranchId(),
        'name' => $this->name,
        'sku' => $this->sku,
        'category' => $this->category,
        'uom' => $this->uom,
        'reorderlevel' => $this->reorderlevel,
        'maxstocklevel' => $this->maxstocklevel,
        'status' => $this->status,
    ];
    // Check if creating or updating
    if ($this->isEditing) {
        // UPDATE ACTION
        if ($this->isSuperAdmin()) {
            $this->executeUpdate('item', $this->itemId, $itemData);
            $this->dispatch('toast', message: 'Item updated successfully', type: 'success');
            $this->resetForm();
        } else {
            // Show reason modal
            $this->auditAction = 'update';
            $this->pendingItemData = $itemData;
            $this->showAuditModal = true;
        }
    } else {
        // CREATE ACTION
        if ($this->isSuperAdmin()) {
            $this->executeCreate('item', $itemData);
            $this->dispatch('toast', message: 'Item created successfully', type: 'success');
            $this->resetForm();
        } else {
            // Show reason modal
            $this->auditAction = 'create';
            $this->pendingItemData = $itemData;
            $this->showAuditModal = true;
        }
    }
}
// New method: Submit for approval
public function submitForAudit()
{
    $this->validate(['auditReason' => 'required|string|min:10|max:500']);
    try {
        $result = $this->submitForApproval(
            $this->auditAction,  // 'create' or 'update'
            'item',              // resource type
            $this->isEditing ? $this->itemId : null,
            $this->pendingItemData,
            $this->auditReason
        );
        $this->dispatch('toast', message: $result['message'], type: 'success');
        $this->resetForm();
    } catch (\Exception $e) {
        $this->dispatch('toast', message: $e->getMessage(), type: 'error');
    }
}
// New method: Delete with approval
public function delete($itemId)
{
    $item = Item::findOrFail($itemId);
    if ($this->isSuperAdmin()) {
        $item->delete();
        $this->dispatch('toast', message: 'Item deleted successfully', type: 'success');
    } else {
        // Show reason modal for deletion
        $this->auditAction = 'delete';
        $this->pendingItemId = $itemId;
        $this->pendingItemData = ['id' => $itemId, 'name' => $item->name];
        $this->showAuditModal = true;
    }
}
// Submit deletion for approval
public function submitDelete()
{
    $this->validate(['auditReason' => 'required|string|min:10|max:500']);
    try {
        $result = $this->submitForApproval(
            'delete',
            'item',
            $this->pendingItemId,
            $this->pendingItemData,
            $this->auditReason
        );
        $this->dispatch('toast', message: $result['message'], type: 'success');
        $this->resetForm();
    } catch (\Exception $e) {
        $this->dispatch('toast', message: $e->getMessage(), type: 'error');
    }
}
private function resetForm()
{
    $this->reset([
        'itemId', 'name', 'sku', 'category', 'uom', 
        'reorderlevel', 'maxstocklevel', 'status',
        'isEditing', 'showModal', 'showAuditModal',
        'auditAction', 'auditReason', 'pendingItemData'
    ]);
}

public function submitStockAdjustment()
{
    $this->validate([
        'quantityavailable' => 'required|numeric|min:0',
        'quantityreserved' => 'required|numeric|min:0',
        'quantitydamaged' => 'required|numeric|min:0',
        'auditReason' => 'required|string|min:10|max:500',
    ]);
    try {
        $stock = Stock::findOrFail($this->editingStockId);
        $adjustmentData = [
            'quantityavailable' => $this->quantityavailable,
            'quantityreserved' => $this->quantityreserved,
            'quantitydamaged' => $this->quantitydamaged,
            'notes' => $this->notes,
        ];
        if ($this->isSuperAdmin()) {
            // Direct execution for super admin
            $this->executeAction('adjust', 'stock', $this->editingStockId, $adjustmentData);
            $this->dispatch('toast', message: 'Stock adjusted successfully', type: 'success');
            $this->resetForm();
        } else {
            // Submit for approval
            $result = $this->submitForApproval(
                'adjust',
                'stock',
                $this->editingStockId,
                $adjustmentData,
                $this->auditReason
            );
            $this->dispatch('toast', message: $result['message'], type: 'success');
            $this->resetForm();
        }
    } catch (\Exception $e) {
        $this->dispatch('toast', message: $e->getMessage(), type: 'error');
    }
}
// Override adjust handler for stock-specific logic
protected function handleAdjust(string $resourceType, ?string $resourceId, array $payload): void
{
    if ($resourceType !== 'stock' || !$resourceId) {
        throw new \Exception("Invalid resource for adjust action");
    }
    $stock = Stock::findOrFail($resourceId);
    // Update stock quantities
    $stock->update([
        'quantityavailable' => $payload['quantityavailable'] ?? $stock->quantityavailable,
        'quantityreserved' => $payload['quantityreserved'] ?? $stock->quantityreserved,
        'quantitydamaged' => $payload['quantitydamaged'] ?? $stock->quantitydamaged,
    ]);
    // Log stock movement
    $oldQuantity = $stock->getOriginal('quantityavailable');
    $newQuantity = $payload['quantityavailable'] ?? $oldQuantity;
    StockMovement::create([
        'stockid' => $stock->id,
        'type' => 'adjustment',
        'quantity' => abs($newQuantity - $oldQuantity),
        'quantitybefore' => $oldQuantity,
        'quantityafter' => $newQuantity,
        'referencetype' => 'approvaladjustment',
        'movedbyid' => $this->getCurrentActor()->id,
        'movedbytype' => getclass($this->getCurrentActor()),
        'notes' => $payload['notes'] ?? 'Approved adjustment',
        'movementdate' => now(),
    ]);
}
private function resetForm()
{
    $this->reset([
        'editingStockId', 'quantityavailable', 'quantityreserved', 
        'quantitydamaged', 'notes', 'showEditModal', 'showAuditModal',
        'auditReason', 'auditAction'
    ]);
}

🔄 How It Integrates with Trait
The Flow
User in Inventory
│
├─ super admin?
│  ├─ YES → Direct execution
│  │      → executeCreate/Update/Delete/Adjust
│  │      → Logged to auditlogs
│  └─ NO  → Reason modal shown
│          → submitForApproval() called
│          → Stored in approvalauditrequests
│          │
│          └─ Admin reviews in InventoryApprovals
│             │
│             └─ Approves
│                └─ executeApprovedAction() called
│                   └─ Trait's executeAction() runs
│                      └─ Action completes
│                         └─ Logged to auditlogs
