CALLBACK SYSTEM - QUICK START GUIDE
=====================================
For developers integrating with the new callback system

BASIC USAGE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CREATE A CALLBACK:

php
// Sales → Production return
$callback = ProductDispatchCallback::create([
    'product_dispatch_id' => $dispatch->id,
    'sales_shift_id' => $shift->id,
    'product_id' => $product->id,
    'quantity' => 10,
    'uom' => 'KG',
    'reason' => 'expired',
    'callback_time' => now(),
]);

// Production → Inventory issue
$callback = ProductionCallback::create([
    'shift_id' => $shift->id,
    'source_type' => 'raw_material_from_stock', // or 'finished_product_reject'
    'item_id' => $item->id, // for raw materials
    'product_id' => $product->id, // for finished products
    'quantity' => 5,
    'uom' => 'KG',
    'reason' => 'damaged',
    'callback_time' => now(),
]);


APPROVE A CALLBACK:

php
// Automatic: uses current_actor() (logged-in User or Employee)
$callback->approve();

// Or explicit: pass specific actor
$callback->approve($employee);


MARK AS RECEIVED (Production callbacks only):

php
$callback->markAsReceived(); // Uses current_actor()


COMPLETE A CALLBACK:

php
// ProductDispatchCallback - updates stock automatically
$callback->completeWithStockUpdate();

// ProductionCallback - stock updated on approve(), then:
$callback->complete();


REJECT A CALLBACK:

php
$callback->reject(reason: 'Product is actually acceptable');

// With current_actor()
$callback->reject(reason: 'We need to investigate further');


GET CALLBACK STATUS:

php
$status = $callback->status; // Returns CallbackStatus enum

// Get human-readable status
echo $status->getLabel(); // "Pending Approval"

// Get badge color for UI
echo $status->getBadgeColor(); // Tailwind CSS classes


QUERY CALLBACKS:

php
// Get pending callbacks
$pending = ProductDispatchCallback::pending()->get();

// Get approved callbacks
$approved = ProductDispatchCallback::approved()->get();

// Get completed callbacks
$completed = ProductDispatchCallback::completed()->get();

// Filter by product
$byProduct = ProductDispatchCallback::byProduct($productId)->get();

// Filter by sales shift
$byShift = ProductDispatchCallback::bySalesShift($shiftId)->get();

// For production callbacks
$rawMaterial = ProductionCallback::rawMaterial()->get();
$finishedProducts = ProductionCallback::finishedProduct()->get();


GET RELATED DATA:

php
// Who recorded this callback?
$actor = $callback->recordedBy; // User or Employee

// Who approved it?
$approver = $callback->approvedBy; // User or Employee

// What product/item?
if ($callback->isRawMaterial()) {
    $item = $callback->item;
} else {
    $product = $callback->product;
}

// Original dispatch (if exists)
$dispatch = $callback->productDispatch; // May be NULL for orphaned callbacks

COMMON WORKFLOWS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

WORKFLOW 1: Sales Returns Product to Production

php
// 1. Sales creates callback
$callback = ProductDispatchCallback::create([
    'product_dispatch_id' => $dispatch->id,
    'sales_shift_id' => $shift->id,
    'product_id' => $product->id,
    'recorded_by_id' => current_actor()->id,
    'recorded_by_type' => get_class(current_actor()),
    'quantity' => 10,
    'reason' => 'damaged',
]);
// Status: pending

// 2. Production approves
$callback->approve(); // Uses current_actor()
// Status: approved_by_production

// 3. Production marks received
$callback->markAsReceived(); // Uses current_actor()
// Status: received_by_production

// 4. Production completes (updates stock)
$callback->completeWithStockUpdate();
// Status: completed
// Stock automatically updated!


WORKFLOW 2: Production Reports Raw Material Damage

php
// 1. Production creates callback
$callback = ProductionCallback::create([
    'shift_id' => $shift->id,
    'source_type' => 'raw_material_from_stock',
    'item_id' => $item->id,
    'quantity' => 5,
    'reason' => 'contamination',
]);
// Status: pending

// 2. Inventory approves (stock updates automatically!)
$callback->approve(); // Uses current_actor()
// Status: approved_by_inventory
// Stock automatically updated!

// 3. Inventory completes
$callback->complete();
// Status: completed


WORKFLOW 3: Production Reports Finished Product Quality Issue

php
// 1. Production creates callback
$callback = ProductionCallback::create([
    'shift_id' => $shift->id,
    'source_type' => 'finished_product_reject',
    'product_id' => $product->id,
    'quantity' => 20,
    'reason' => 'quality_issue',
]);
// Status: pending

// 2. Inventory approves (stock updates automatically!)
$callback->approve(); // Uses current_actor()
// Status: approved_by_inventory
// DailyProduce automatically updated!

// 3. Inventory completes
$callback->complete();
// Status: completed


WORKFLOW 4: Inventory Rejects Callback

php
// 1. Callback created
$callback = ProductionCallback::create([...]);
// Status: pending

// 2. Inventory rejects
$callback->reject(reason: 'Quality standards actually met');
// Status: rejected
// Reason appended to notes

VALIDATION & ERROR HANDLING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

VALIDATION HAPPENS AUTOMATICALLY:

php
// This will throw ValidationException if qty > available
$callback = ProductDispatchCallback::create([
    'product_dispatch_id' => $dispatch->id,
    'quantity' => 1000, // More than received!
    ...
]);
// Exception: "Callback quantity (1000) exceeds available (100)"


HANDLE ERRORS PROPERLY:

php
try {
    $callback = ProductDispatchCallback::create($data);
} catch (ValidationException $e) {
    // Quantity validation failed
    return back()->with('error', $e->getMessage());
} catch (RuntimeException $e) {
    // Related data not found
    return back()->with('error', 'Product dispatch not found');
} catch (Exception $e) {
    // Other error
    Log::error('Callback creation failed', ['error' => $e->getMessage()]);
    return back()->with('error', 'Failed to create callback');
}


TRANSACTION SAFETY:

php
// completeWithStockUpdate() is atomic
try {
    $callback->completeWithStockUpdate();
    // Either fully completes with stock update, or rolls back everything
} catch (Exception $e) {
    // All changes rolled back
    Log::error('Completion failed', ['callback' => $callback->id]);
}

TESTING EXAMPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

CREATE AND COMPLETE CALLBACK:

php
// tests/Feature/Callbacks/CompleteTest.php
public function test_complete_callback_updates_stock()
{
    $dispatch = ProductDispatch::factory()->create();
    $callback = ProductDispatchCallback::factory()
        ->for($dispatch)
        ->create(['status' => 'received_by_production']);
    
    // Initial stock
    $stock = ProductStock::where('product_id', $callback->product_id)
        ->where('sales_shift_id', $callback->sales_shift_id)
        ->first();
    $originalQty = $stock->callback_quantity;
    
    // Complete with stock update
    $callback->completeWithStockUpdate();
    
    // Verify stock updated
    $stock->refresh();
    $this->assertEquals(
        $originalQty + $callback->quantity,
        $stock->callback_quantity
    );
    
    // Verify callback completed
    $this->assertEquals('completed', $callback->status->value);
}


VALIDATE QUANTITY:

php
public function test_validate_quantity_prevents_over_return()
{
    $dispatch = ProductDispatch::factory(['received_quantity' => 100])->create();
    
    // First callback: 50 units (OK)
    $callback1 = ProductDispatchCallback::factory()
        ->for($dispatch)
        ->create(['quantity' => 50]);
    
    // Second callback: 60 units (exceeds available 50) - should fail
    $this->expectException(ValidationException::class);
    
    ProductDispatchCallback::create([
        'product_dispatch_id' => $dispatch->id,
        'quantity' => 60, // Exceeds available (100 - 50)
    ]);
}


APPROVE WITH AUTO STOCK UPDATE:

php
public function test_production_callback_approve_updates_stock()
{
    $callback = ProductionCallback::factory()
        ->state(['source_type' => 'raw_material_from_stock'])
        ->create();
    
    $stock = Stock::where('item_id', $callback->item_id)->first();
    $originalAvailable = $stock->quantity_available;
    
    // Approve (stock updates automatically)
    $callback->approve();
    
    // Verify stock decreased
    $stock->refresh();
    $this->assertEquals(
        $originalAvailable - $callback->quantity,
        $stock->quantity_available
    );
    
    // Verify status updated
    $this->assertEquals('approved_by_inventory', $callback->status->value);
}

TROUBLESHOOTING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Issue: "No authenticated actor found"
Solution: Make sure user is logged in (via auth() or auth('employees'))
Check: echo current_actor(); // Should return User or Employee

Issue: "Product stock record not found"
Solution: Ensure ProductStock record exists for the product/shift combo
Check: ProductStock::where('product_id', $id)->where('sales_shift_id', $id)->exists();

Issue: "Recipe not found for product"
Solution: Ensure Recipe exists for the product
Check: Recipe::where('product_id', $id)->first();

Issue: "ValidationException: quantity exceeds available"
Solution: Check sum of all existing callbacks for the dispatch
Check: ProductDispatchCallback::where('product_dispatch_id', $id)->sum('quantity');

Issue: "Stock not updated after approve()"
Solution: For ProductDispatchCallback, use completeWithStockUpdate() not just approve()
For ProductionCallback, stock updates on approve() - verify Stock record exists

USE EAGER LOADING:

php
// BAD - N+1 queries
$callbacks = ProductDispatchCallback::all();
foreach ($callbacks as $callback) {
    echo $callback->product->name; // Query for each row
}

// GOOD - Eager loaded
$callbacks = ProductDispatchCallback::with('product')->get();
foreach ($callbacks as $callback) {
    echo $callback->product->name; // No additional queries
}


USE INDEXES FOR FILTERING:

php
// FAST - Uses (status, callback_time) index
$callbacks = ProductDispatchCallback::where('status', 'pending')
    ->where('callback_time', '>=', now()->subDay())
    ->get();

// SLOW - Full table scan
$callbacks = ProductDispatchCallback::where('quantity', '>', 50)->get();


PAGINATE LARGE RESULTS:

php
// GOOD - Pagination
$callbacks = ProductDispatchCallback::paginate(50);

// BAD - Loading all at once
$callbacks = ProductDispatchCallback::get(); // Could be thousands


USE TRANSACTIONS FOR BATCH OPERATIONS:

php
DB::transaction(function () {
    foreach ($callbackIds as $id) {
        $callback = ProductDispatchCallback::find($id);
        $callback->approve();
    }
});
